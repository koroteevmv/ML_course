Скрипт ниже создаёт три файла в текущей директории. Запускать достаточно один раз.

```python
# prepare_data.py
import numpy as np
import pandas as pd


def generate_basic(n=200, seed=42, path='multireg_basic.csv'):
    """Четыре признака, линейная зависимость + шум."""
    rng = np.random.RandomState(seed)
    x1 = rng.uniform(-3, 3, n)
    x2 = rng.uniform(0, 5, n)
    x3 = rng.normal(0, 1, n)
    x4 = rng.uniform(10, 20, n)
    # Истинная зависимость: y = 3 + 2.5*x1 - 1.3*x2 + 0.8*x3 + 0.5*x4 + шум
    y = 3.0 + 2.5 * x1 - 1.3 * x2 + 0.8 * x3 + 0.5 * x4 + rng.randn(n) * 0.5
    df = pd.DataFrame({'x1': x1, 'x2': x2, 'x3': x3, 'x4': x4, 'y': y})
    df.to_csv(path, index=False)
    print(f'{path}: {df.shape}')
    return df


def generate_poly(n=150, seed=0, path='multireg_poly.csv'):
    """Один признак, квадратичная зависимость + шум."""
    rng = np.random.RandomState(seed)
    x = rng.uniform(-3, 3, n)
    # Истинная зависимость: y = 1.5 - 0.5*x + 0.8*x^2 + шум
    y = 1.5 - 0.5 * x + 0.8 * x ** 2 + rng.randn(n) * 0.5
    df = pd.DataFrame({'x': x, 'y': y})
    df.to_csv(path, index=False)
    print(f'{path}: {df.shape}')
    return df


def generate_nonlinear(n=120, seed=1, path='multireg_nonlinear.csv'):
    """Один признак и пять разных нелинейных зависимостей."""
    rng = np.random.RandomState(seed)
    # x в диапазоне [0.3, 4.5] — удобно для всех функций
    x = np.linspace(0.3, 4.5, n)
    df = pd.DataFrame({
        'x': x,
        'y_sqrt': 1.0 + 2.0 * np.sqrt(x)          + rng.randn(n) * 0.15,
        'y_log':  0.5 + 2.0 * np.log(x)           + rng.randn(n) * 0.15,
        'y_inv':  1.5 + 3.0 / x                   + rng.randn(n) * 0.10,
        'y_sin':  0.0 + 2.5 * np.sin(x)           + rng.randn(n) * 0.20,
    })
    # Отдельно экспонента — на узком диапазоне, чтобы значения не «взлетали»
    x_exp = np.linspace(0.3, 2.0, n)
    df_exp = pd.DataFrame({
        'x': x_exp,
        'y_exp': 0.5 + 1.5 * np.exp(x_exp) + rng.randn(n) * 0.20,
    })
    df_exp.to_csv('multireg_exp.csv', index=False)
    df.to_csv(path, index=False)
    print(f'{path}: {df.shape}')
    print(f'multireg_exp.csv: {df_exp.shape}')
    return df


if __name__ == '__main__':
    generate_basic()
    generate_poly()
    generate_nonlinear()
```

#### Методические указания

##### Постановка задачи множественной регрессии

**Множественная линейная регрессия** предполагает, что целевая переменная $y$ линейно зависит от нескольких признаков $x_1, x_2, \dots, x_p$:

$$y = \beta_0 + \beta_1 x_1 + \beta_2 x_2 + \dots + \beta_p x_p + \varepsilon$$

где $\beta_0$ — свободный коэффициент (intercept), $\beta_1, \dots, \beta_p$ — коэффициенты при признаках, $\varepsilon$ — случайная ошибка.

Задача — по обучающей выборке $(X, y)$ найти такие $\beta$, которые минимизируют **сумму квадратов остатков**:

$$\text{MSE}(\beta) = \frac{1}{n}\sum_{i=1}^{n} \left(y_i - \hat{y}_i\right)^2, \qquad \hat{y}_i = \beta_0 + \sum_{j=1}^{p} \beta_j x_{ij}$$

Формально это задача оптимизации, но у неё есть **аналитическое решение** — нормальное уравнение, которое мы сейчас и реализуем.

> **Чем множественная регрессия отличается от парной?** В парной регрессии один признак и один коэффициент $\beta_1$. В множественной — несколько признаков, и коэффициенты показывают вклад каждого признака **при фиксированных остальных**. Это позволяет учесть совместное влияние и «очистить» эффект каждого признака от корреляций с другими.

##### Нормальное уравнение

Добавим к матрице признаков столбец единиц (чтобы $\beta_0$ тоже вошёл в формулу):

$$X_{\text{design}} = \begin{bmatrix} 1 & x_{1,1} & \dots & x_{1,p} \\ 1 & x_{2,1} & \dots & x_{2,p} \\ \vdots & \vdots & \ddots & \vdots \\ 1 & x_{n,1} & \dots & x_{n,p} \end{bmatrix}, \quad \beta = \begin{bmatrix} \beta_0 \\ \beta_1 \\ \vdots \\ \beta_p \end{bmatrix}$$

Тогда предсказание — это матричное произведение $\hat{y} = X_{\text{design}} \beta$, а оптимальные коэффициенты находятся по формуле:

$$\hat{\beta} = \left(X_{\text{design}}^T X_{\text{design}}\right)^{-1} X_{\text{design}}^T y$$

Вывод этой формулы мы опускаем (это задача оптимизации квадратичной функции), но важно понимать: **это точное решение**, а не итеративное, как в градиентном спуске. Для небольших и средних задач $(n \sim 10^4, p \sim 10^2)$ его достаточно.

> **Численная устойчивость.** На практике вместо `inv(X.T @ X) @ X.T @ y` лучше использовать `np.linalg.lstsq(X, y)`. Оно даёт тот же результат, но устойчивее к плохо обусловленным матрицам и мультиколлинеарности. Мы покажем оба варианта.

##### Построение модели руками

Реализуем два метода — `fit` и `predict` — в виде функций:

```python
def fit_linear_regression(X, y):
    """Возвращает коэффициенты beta (включая intercept в beta[0])."""
    n = len(X)
    X_design = np.hstack([np.ones((n, 1)), X])
    # Нормальное уравнение
    beta = np.linalg.inv(X_design.T @ X_design) @ X_design.T @ y
    return beta


def predict_linear(X, beta):
    """Предсказание по матрице признаков и коэффициентам."""
    n = len(X)
    X_design = np.hstack([np.ones((n, 1)), X])
    return X_design @ beta
```

Обучим на train и сделаем предсказание на test:

```python
beta_manual = fit_linear_regression(X_tr, y_tr)
y_pred_manual = predict_linear(X_te, beta_manual)

print('Коэффициенты (руками):')
print(f'  intercept = {beta_manual[0]:+.3f}')
for name, b in zip(['x1', 'x2', 'x3', 'x4'], beta_manual[1:]):
    print(f'  {name}: {b:+.3f}')
```

У вас должно получиться что-то вроде:

```
intercept = +3.05
x1: +2.48
x2: -1.31
x3: +0.79
x4: +0.50
```

Истинные коэффициенты, которые мы «закладывали» при генерации: `3.0, 2.5, -1.3, 0.8, 0.5`. Модель их восстановила с точностью до десятых. Небольшое расхождение — из-за шума.

**Метрика качества.** Основная метрика — **коэффициент детерминации** $R^2$:

$$R^2 = 1 - \frac{\sum_i (y_i - \hat{y}_i)^2}{\sum_i (y_i - \bar{y})^2}$$

$R^2 = 1$ — идеальное совпадение, $R^2 = 0$ — модель не лучше, чем предсказание среднего. Считаем:

```python
from sklearn.metrics import mean_squared_error, r2_score

mse_manual = mean_squared_error(y_te, y_pred_manual)
r2_manual = r2_score(y_te, y_pred_manual)
print(f'MSE (руками): {mse_manual:.3f}, R² (руками): {r2_manual:.3f}')
```

##### Библиотечная реализация

`sklearn.linear_model.LinearRegression` делает то же самое, но с дополнительными удобствами (численная устойчивость, работа с DataFrame, атрибуты `coef_` и `intercept_`):

```python
from sklearn.linear_model import LinearRegression

model = LinearRegression().fit(X_tr, y_tr)
y_pred_sklearn = model.predict(X_te)

print('Коэффициенты (sklearn):')
print(f'  intercept = {model.intercept_:+.3f}')
for name, b in zip(['x1', 'x2', 'x3', 'x4'], model.coef_):
    print(f'  {name}: {b:+.3f}')

mse_sklearn = mean_squared_error(y_te, y_pred_sklearn)
r2_sklearn = r2_score(y_te, y_pred_sklearn)
print(f'MSE (sklearn): {mse_sklearn:.3f}, R² (sklearn): {r2_sklearn:.3f}')
```

##### Сравнение результатов

Коэффициенты должны совпасть с точностью до численной погрешности:

```python
print('Сравнение коэффициентов:')
print(f'  intercept: manual={beta_manual[0]:+.6f}, sklearn={model.intercept_:+.6f}, '
      f'разница={abs(beta_manual[0] - model.intercept_):.2e}')
for i, name in enumerate(['x1', 'x2', 'x3', 'x4']):
    diff = abs(beta_manual[i+1] - model.coef_[i])
    print(f'  {name}: manual={beta_manual[i+1]:+.6f}, sklearn={model.coef_[i]:+.6f}, '
          f'разница={diff:.2e}')

print(f'\nMSE: manual={mse_manual:.6f}, sklearn={mse_sklearn:.6f}')
print(f'R²:  manual={r2_manual:.6f}, sklearn={r2_sklearn:.6f}')
```

Разница должна быть порядка $10^{-14}$ — это просто накопленная погрешность вычислений с плавающей точкой. Если вы получили что-то больше — проверьте, что `X_design` строится одинаково.

**Замечание о численной устойчивости.** Если признаки сильно коррелированы или у них разный масштаб, `np.linalg.inv(X.T @ X)` может дать большую погрешность. Проверим это специально:

```python
# Добавим признак, который почти повторяет x1 (сильная корреляция)
X_tr_collinear = np.hstack([X_tr, X_tr[:, [0]] + np.random.randn(len(X_tr), 1) * 0.01])
X_te_collinear = np.hstack([X_te, X_te[:, [0]] + np.random.randn(len(X_te), 1) * 0.01])

# Через inv
beta_inv = fit_linear_regression(X_tr_collinear, y_tr)

# Через lstsq (устойчивый метод)
X_design = np.hstack([np.ones((len(X_tr_collinear), 1)), X_tr_collinear])
beta_lstsq, *_ = np.linalg.lstsq(X_design, y_tr, rcond=None)

print('Через inv:  ', beta_inv.round(3))
print('Через lstsq:', beta_lstsq.round(3))
```

Коэффициенты будут заметно различаться, при этом предсказания — почти совпадать. Это классическая проблема **мультиколлинеарности**: коэффициенты становятся нестабильными, хотя сама модель предсказывает нормально. На практике всегда используйте `np.linalg.lstsq` (или `sklearn`, который внутри использует `scipy.linalg.lstsq`).

**Как читать эти графики:**

- На первом графике точки должны лежать около красной диагонали. Чем ближе — тем лучше. Систематические отклонения (например, точки «загибаются» вверх при больших `y`) означают, что **линейная модель не описывает зависимость** — возможно, нужны нелинейные признаки (см. задания).
- На втором графике остатки должны быть **равномерно разбросаны** вокруг нуля без видимых паттернов. Если виден «конус» (разброс растёт с ростом ŷ) — есть гетероскедастичность; если видна кривая — модель недоописывает нелинейность.
