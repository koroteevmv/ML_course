##### Подготовка данных

Для частей B–D понадобятся два CSV-файла. Их можно сгенерировать приведённым ниже кодом (файлы сохраняются в текущую директорию). Если преподаватель предоставил файлы, этот шаг можно пропустить.

```python
import numpy as np
import pandas as pd

rng = np.random.RandomState(42)

# --- groups.csv ---
groups = pd.DataFrame({
    'group':      ['G-101', 'G-102', 'G-201', 'G-202', 'G-301'],
    'faculty':    ['ФПМИ', 'ФПМИ', 'ФИТ', 'ФИТ', 'ФЭТ'],
    'course':     [1, 1, 2, 2, 3],
    'head':       ['Иванов А.', 'Петрова М.', 'Сидоров В.', 'Кузнецова Е.', 'Орлов Д.'],
})
groups.to_csv('groups.csv', index=False)

# --- students.csv ---
n = 120
groups_list = groups['group'].tolist()
faculties = ['Москва', 'Санкт-Петербург', 'Новосибирск', 'Казань', 'Екатеринбург']

students = pd.DataFrame({
    'student_id':      np.arange(1, n + 1),
    'name':            [f'Студент_{i:03d}' for i in range(1, n + 1)],
    'age':             rng.choice([17, 18, 19, 20, 21, 22, 23, 25, 30], n,
                                  p=[0.05, 0.15, 0.2, 0.2, 0.15, 0.1, 0.08, 0.05, 0.02]),
    'group':           rng.choice(groups_list, n),
    'math_score':      np.clip(rng.normal(65, 15, n), 0, 100).round(1),
    'physics_score':   np.clip(rng.normal(60, 18, n), 0, 100).round(1),
    'cs_score':        np.clip(rng.normal(70, 12, n), 0, 100).round(1),
    'enrollment_date': pd.to_datetime('2023-09-01') +
                       pd.to_timedelta(rng.randint(0, 30, n), unit='D'),
    'city':            rng.choice(faculties, n),
})

# Внесём пропуски
idx_age = rng.choice(n, size=8, replace=False)
students.loc[idx_age, 'age'] = np.nan
idx_math = rng.choice(n, size=6, replace=False)
students.loc[idx_math, 'math_score'] = np.nan
idx_phys = rng.choice(n, size=4, replace=False)
students.loc[idx_phys, 'physics_score'] = np.nan

students['enrollment_date'] = students['enrollment_date'].dt.strftime('%Y-%m-%d')
students.to_csv('students.csv', index=False)

print('Файлы students.csv и groups.csv созданы.')
print(students.shape, groups.shape)
```

Если преподаватель предоставил файлы, этот шаг можно пропустить и считать данные из csv-файла:
